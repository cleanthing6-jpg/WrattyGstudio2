            t[_k] = _v

        for _k, _v in (_PRESET.get("role_treat_backing") or {}).items():

        for _k, _v in (_PRESET.get("role_treat_adlib") or {}).items():
            t[_k] = _v

        for _k, _v in (_PRESET.get(f"role_treat_{role}") or {}).items():
            t[_k] = _v

            t[_k] = t.get(_k, 0.0) + float(_v)
    return t

def _role_buses(pre, sr, bpm, raised, beat=None, fx_plans=None):
    """Give each vocal role its OWN glue and its OWN reverb depth, then sum.
    No cross-role compression: the lead is never glued to backing or adlib.
    Per-bus peak normalisation is SKIPPED (headroom=False) so the locked role
    levels are untouched; the summed bus normalises once at the end."""
    out = []
    first = None
    for r, y0 in pre.items():
        y = np.asarray(y0, dtype=np.float32)
        if polish is not None:
            try:
                _y, _ = polish.dynamic_eq(y, sr, get_stats=True)
                if _y.shape == y.shape and np.isfinite(_y).all():
                    y = np.asarray(_y, dtype=np.float32)
            except Exception:
                pass
        if raised:
            y = (y * (10.0 ** (raised / 20.0))).astype(np.float32)
        cfg = ROLE_BUS.get(r) or {"glue": None, "plate": 1.0, "delay": 1.0}
        if r == "response":
            cfg = {"glue": None, "plate": 0.0, "delay": 0.0}  # NO reverb/delay for punch-ins
        elif r in ("intro", "outro"):
            cfg = {"glue": None, "plate": 1.5, "delay": 1.5}  # Heavy reverb/delay for intro/outro
        if cfg.get("glue"):
            _thr, _rat = cfg["glue"]
            y = _glue(y, sr, thr=_thr, ratio=_rat)
        if cfg.get("exciter"):
            y = _exciter(y, sr)
        _pl = float(cfg.get("plate", cfg.get("space", 1.0)))
        _dl = float(cfg.get("delay", _pl))
        if fx_plans and r in fx_plans:
            _pl *= fx_plans[r].get("reverb_send_scale", 1.0)
            _dl *= fx_plans[r].get("delay_send_scale", 1.0)
        _pfd = bool(cfg.get("plate_from_delay", False))
        if _pl or _dl or _pfd:
            _nt = float(cfg.get("note", 0.75))
            # TRY NEW vocal_space PATCH
            try:
                import vocal_space
                _bus, _vs_rep = vocal_space.build(y, sr, bpm=bpm, genre=("rap" if r == "feature" else _PRESET.get("_name", "afrobeats")), role=("lead" if r == "feature" else r), beat=beat, space=_pl)
                y = y + _bus  # Additive bus: dry vocal + wet space
                _ms, _kind = 0, "vocal_space"
                _PRESET["_vocal_space_used"] = True
                if "vocal_space_rep" not in _PRESET: _PRESET["vocal_space_rep"] = {}
                _PRESET["vocal_space_rep"][r] = _vs_rep
            except Exception as _vs_err:
                import traceback
                _PRESET["_vocal_space_error"] = traceback.format_exc()[-300:]
                # Fallback to old _ambience if vocal_space crashes
                y, _ms, _kind = _ambience(y, sr, bpm, scale=_pl,
                                      plate_scale=_pl, delay_scale=_dl,
                                      plate_from_delay=_pfd, headroom=False,
                                      note_mult=_nt, role=r)
            if first is None:
                first = (_ms, _kind)
        out.append(y)
    return _sum(out), first
